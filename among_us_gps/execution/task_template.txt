def _client_rect(hwnd):
    try:
        rect  = _win32gui.GetWindowRect(hwnd)
        crect = _win32gui.GetClientRect(hwnd)
        bw = int((rect[2] - rect[0] - crect[2]) / 2)
        th = int(rect[3] - rect[1] - crect[3] - bw)
        return rect[0] + bw, rect[1] + th, crect[2], crect[3]
    except Exception:
        return None


def _point_in_polygon(px, py, poly):
    n = len(poly)
    if n < 3:
        return False
    inside = False
    j = n - 1
    for i in range(n):
        xi, yi = poly[i]; xj, yj = poly[j]
        if ((yi > py) != (yj > py)) and \
           (px < (xj - xi) * (py - yi) / (yj - yi + 1e-12) + xi):
            inside = not inside
        j = i
    return inside


def _random_in_rect(rect):
    x1, y1, x2, y2 = rect
    xa, xb = min(x1, x2), max(x1, x2)
    ya, yb = min(y1, y2), max(y1, y2)
    return _random.uniform(xa, xb), _random.uniform(ya, yb)


def _random_in_poly(poly):
    if not poly:
        return 0.5, 0.5
    xs = [p[0] for p in poly]; ys = [p[1] for p in poly]
    x1, x2 = min(xs), max(xs); y1, y2 = min(ys), max(ys)
    for _ in range(40):
        rx = _random.uniform(x1, x2); ry = _random.uniform(y1, y2)
        if _point_in_polygon(rx, ry, poly):
            return rx, ry
    return sum(xs)/len(xs), sum(ys)/len(ys)


def _click_hold(x, y, durata):
    _pag.moveTo(x, y, duration=0.1)
    if durata > 0:
        _pag.mouseDown(); _time.sleep(durata); _pag.mouseUp()
    else:
        _pag.click()


def _drag_umano(sx, sy, ex, ey, durata):
    _pag.moveTo(sx, sy, duration=_random.uniform(0.15, 0.25),
                tween=_pag.easeOutQuad)
    _pag.mouseDown(button='left')
    _time.sleep(_random.uniform(0.05, 0.1))
    if durata > 0:
        steps = max(10, int(durata * 60)); st = durata / steps
        off = _random.randint(-40, 40)
        mx = (sx + ex) / 2 + off; my = (sy + ey) / 2 + off
        for i in range(1, steps + 1):
            t = i / steps; te = -(_math.cos(_math.pi * t) - 1) / 2
            _pag.moveTo(int((1-te)**2*sx + 2*(1-te)*te*mx + te**2*ex),
                        int((1-te)**2*sy + 2*(1-te)*te*my + te**2*ey))
            _time.sleep(st)
        _pag.moveTo(ex, ey)
    else:
        _pag.moveTo(ex, ey)
    _time.sleep(_random.uniform(0.05, 0.15))
    _pag.mouseUp(button='left')


def _drag_multi(punti_abs, durata_totale):
    if len(punti_abs) < 2:
        return
    sx, sy = punti_abs[0]
    _pag.moveTo(sx, sy, duration=_random.uniform(0.15, 0.25),
                tween=_pag.easeOutQuad)
    _pag.mouseDown(button='left')
    _time.sleep(_random.uniform(0.05, 0.1))
    total = sum(_math.hypot(punti_abs[i+1][0]-punti_abs[i][0],
                             punti_abs[i+1][1]-punti_abs[i][1])
                for i in range(len(punti_abs)-1))
    if total <= 0:
        _pag.mouseUp(button='left'); return
    for i in range(len(punti_abs)-1):
        ax, ay = punti_abs[i]; bx, by = punti_abs[i+1]
        seg = _math.hypot(bx-ax, by-ay)
        dur_seg = max(0.05, durata_totale * (seg/total))
        steps = max(6, int(dur_seg * 60)); st = dur_seg / steps
        off = _random.randint(-3, 3)
        mx = (ax+bx)/2+off; my = (ay+by)/2+off
        for k in range(1, steps+1):
            t = k/steps; te = -(_math.cos(_math.pi*t)-1)/2
            _pag.moveTo(int((1-te)**2*ax + 2*(1-te)*te*mx + te**2*bx),
                        int((1-te)**2*ay + 2*(1-te)*te*my + te**2*by))
            _time.sleep(st)
        _pag.moveTo(bx, by)
    _time.sleep(_random.uniform(0.05, 0.15))
    _pag.mouseUp(button='left')


def _extract_pure_shape(roi_image):
    import cv2
    import numpy as np
    gray = cv2.cvtColor(roi_image, cv2.COLOR_BGRA2GRAY) if roi_image.shape[2] == 4 else cv2.cvtColor(roi_image, cv2.COLOR_BGR2GRAY)
    _, thresh = cv2.threshold(gray, 120, 255, cv2.THRESH_BINARY_INV)
    points = cv2.findNonZero(thresh)
    if points is not None:
        tx, ty, tw, th = cv2.boundingRect(points)
        cropped = thresh[ty:ty+th, tx:tx+tw]
        size = max(tw, th)
        top = (size - th) // 2
        bottom = size - th - top
        left = (size - tw) // 2
        right = size - tw - left
        square = cv2.copyMakeBorder(cropped, top, bottom, left, right, cv2.BORDER_CONSTANT, value=0)
        res = cv2.resize(square, (40, 40), interpolation=cv2.INTER_AREA)
    else:
        res = cv2.resize(thresh, (40, 40))
    return cv2.GaussianBlur(res, (3, 3), 0)


def _drag_seq_tappe(punti_abs, durata_segmento):
    if len(punti_abs) < 2:
        return
    sx, sy = punti_abs[0]
    _pag.moveTo(sx, sy, duration=_random.uniform(0.15, 0.25), tween=_pag.easeOutQuad)
    _pag.mouseDown(button='left')
    _time.sleep(_random.uniform(0.05, 0.1))

    dur_seg = max(0.25, durata_segmento)
    for i in range(1, len(punti_abs)):
        bx, by = punti_abs[i]
        _pag.moveTo(bx, by, duration=dur_seg, tween=_pag.easeInOutQuad)
        _time.sleep(0.15)

    _time.sleep(_random.uniform(0.05, 0.15))
    _pag.mouseUp(button='left')

def _drag_e_tieni(sx, sy, ex, ey, durata, hold):
    _drag_umano(sx, sy, ex, ey, durata)
    if hold > 0:
        _pag.mouseDown(); _time.sleep(hold); _pag.mouseUp()


def esegui_azioni(azioni, hwnd, current_step=0, is_test=False):
    if not _OK:
        return False

    chunks = []
    current_chunk = []
    cooldowns = []
    for az in azioni:
        if az.get('tipo') == 'cooldown':
            chunks.append(current_chunk)
            cooldowns.append(az.get('durata', 0.0))
            current_chunk = []
        else:
            current_chunk.append(az)
    chunks.append(current_chunk)

    if current_step >= len(chunks):
        current_step = max(0, len(chunks) - 1)

    azioni_da_eseguire = chunks[current_step] if not is_test else azioni

    for az in azioni_da_eseguire:
        if _win32gui and hwnd and _win32gui.GetForegroundWindow() != hwnd:
            print("[Task] Gioco non in primo piano, in pausa...", flush=True)
            while _win32gui.GetForegroundWindow() != hwnd:
                _time.sleep(0.5)
            _time.sleep(0.3)

        rect = _client_rect(hwnd)
        if not rect:
            return False
        cx, cy, cw, ch = rect
        tipo   = az.get('tipo')
        durata = az.get('durata', 0.0)
        attesa = az.get('attesa', 0.0)

        if tipo == 'cooldown':
            if is_test:
                print(f"[Test] Attesa cooldown di {durata}s...")
                _time.sleep(durata)
            continue

        if tipo == 'click':
            _click_hold(cx + int(az['rx']*cw), cy + int(az['ry']*ch), durata)
        elif tipo == 'click_rect':
            rx, ry = _random_in_rect(az['rect'])
            _click_hold(cx + int(rx*cw), cy + int(ry*ch), durata)
        elif tipo == 'click_poly':
            rx, ry = _random_in_poly(az['poly'])
            _click_hold(cx + int(rx*cw), cy + int(ry*ch), durata)
        elif tipo == 'drag':
            sx = cx+int(az['start_rx']*cw); sy = cy+int(az['start_ry']*ch)
            ex = cx+int(az['end_rx']*cw);   ey = cy+int(az['end_ry']*ch)
            d  = _math.hypot(az['end_rx']-az['start_rx'], az['end_ry']-az['start_ry'])
            _drag_umano(sx, sy, ex, ey, max(0.05, d*durata))
        elif tipo == 'drag_multi':
            pts = [(cx+int(p[0]*cw), cy+int(p[1]*ch)) for p in az['punti']]
            rel = az['punti']
            tot = sum(_math.hypot(rel[i+1][0]-rel[i][0], rel[i+1][1]-rel[i][1])
                      for i in range(len(rel)-1))
            _drag_multi(pts, max(0.1, tot*durata))
        elif tipo == 'drag_zone':
            sxr, syr = _random_in_poly(az['zone_a'])
            exr, eyr = _random_in_poly(az['zone_b'])
            d = _math.hypot(exr-sxr, eyr-syr)
            _drag_umano(cx+int(sxr*cw), cy+int(syr*ch),
                        cx+int(exr*cw), cy+int(eyr*ch), max(0.05, d*durata))
        elif tipo == 'drag_hold':
            sx = cx+int(az['start_rx']*cw); sy = cy+int(az['start_ry']*ch)
            ex = cx+int(az['end_rx']*cw);   ey = cy+int(az['end_ry']*ch)
            d  = _math.hypot(az['end_rx']-az['start_rx'], az['end_ry']-az['start_ry'])
            _drag_e_tieni(sx, sy, ex, ey, max(0.05, d*durata), az.get('hold', 0.0))
        elif tipo == 'wiring':
            left_pts = az.get('left', [])
            right_pts = az.get('right', [])
            light_pts = az.get('lights', [])
            disp_dx = list(range(len(right_pts)))
            for i in range(len(left_pts)):
                start_rx, start_ry = left_pts[i]
                sx = cx + int(start_rx * cw); sy = cy + int(start_ry * ch)
                
                try:
                    col_sx = _pag.pixel(sx, sy)
                except Exception:
                    col_sx = (0, 0, 0)
                    
                target_scelti = []
                for r_idx in list(disp_dx):
                    end_rx, end_ry = right_pts[r_idx]
                    ex = cx + int(end_rx * cw); ey = cy + int(end_ry * ch)
                    try:
                        col_dx = _pag.pixel(ex, ey)
                        diff = abs(col_sx[0]-col_dx[0]) + abs(col_sx[1]-col_dx[1]) + abs(col_sx[2]-col_dx[2])
                    except Exception:
                        col_dx = (0, 0, 0)
                        diff = 999
                    target_scelti.append((diff, r_idx))
                    print(f"[Wiring] Cavo sx {i} {col_sx} vs Target dx {r_idx} {col_dx} -> diff: {diff}")
                target_scelti.sort(key=lambda item: item[0])
                ordine_tentativi = [t[1] for t in target_scelti]

                _pag.moveTo(sx, sy, duration=0.15)
                _pag.mouseDown(button='left')
                _time.sleep(0.1)
                for r_idx in ordine_tentativi:
                    end_rx, end_ry = right_pts[r_idx]
                    ex = cx + int(end_rx * cw); ey = cy + int(end_ry * ch)
                    light_rx, light_ry = light_pts[r_idx]
                    lx = cx + int(light_rx * cw); ly = cy + int(light_ry * ch)
                    _pag.moveTo(ex, ey, duration=0.25)
                    _time.sleep(0.1)  # Attendi frame per accensione luce
                    try:
                        r, g, b = _pag.pixel(lx, ly)
                        if r > 160 and g > 150 and b < 100:  # Luce gialla accesa
                            disp_dx.remove(r_idx)
                            break
                    except Exception:
                        pass
                _pag.mouseUp(button='left')
                _time.sleep(attesa)
        elif tipo == 'sync_click':
            punti_sync = az.get('punti', [])
            i_sync = 0
            max_retries = 3
            retry_count = 0
            
            check_coords_abs = []
            for cx_rel, cy_rel, _, _ in punti_sync:
                check_coords_abs.append((cx + int(cx_rel * cw), cy + int(cy_rel * ch)))

            while i_sync < len(punti_sync):
                if retry_count >= max_retries:
                    print(f"[sync_click] Tentativi massimi ({max_retries}) raggiunti. Annullamento.")
                    break

                _, _, bx_rel, by_rel = punti_sync[i_sync]
                chk_x, chk_y = check_coords_abs[i_sync]
                btn_x = cx + int(bx_rel * cw)
                btn_y = cy + int(by_rel * ch)
                
                _pag.moveTo(btn_x, btn_y, duration=0.1)
                
                try:
                    base_col = _pag.pixel(chk_x, chk_y)
                except Exception:
                    base_col = (0, 0, 0)
                
                start_t = _time.time()
                clicked = False
                while _time.time() - start_t < 8.0:
                    try:
                        r, g, b = _pag.pixel(chk_x, chk_y)
                        if (r + g + b) > 60 and (abs(r - base_col[0]) + abs(g - base_col[1]) + abs(b - base_col[2])) > 40:
                            _pag.click()
                            clicked = True
                            break
                    except Exception:
                        pass
                    _time.sleep(0.005)
                
                if not clicked:
                    print(f"[sync_click] Timeout al punto {i_sync+1}. Riavvio sequenza.")
                    i_sync = 0
                    retry_count += 1
                    _time.sleep(1.0)
                    continue
                
                _time.sleep(0.2) 

                sequence_failed = False
                failed_at_step = -1
                for k in range(i_sync + 1):
                    prev_chk_x, prev_chk_y = check_coords_abs[k]
                    try:
                        r, g, b = _pag.pixel(prev_chk_x, prev_chk_y)

                            sequence_failed = True
                            failed_at_step = k + 1
                            break 
                    except Exception:
                        sequence_failed = True
                        failed_at_step = k + 1
                        break
                
                if sequence_failed:
                    print(f"[sync_click] Fallimento rilevato al passo {failed_at_step}. Riavvio (tentativo {retry_count + 1}/{max_retries}).")
                    i_sync = 0
                    retry_count += 1
                    _time.sleep(0.75)
                    continue
                    
                i_sync += 1
            _time.sleep(attesa)
        elif tipo == 'click_anomaly':
            _time.sleep(1.0)

            btn_pts = []
            for p in az.get('punti', []):
                if len(p) >= 4:

                else:
                    btn_pts.append((cx + int(p[0]*cw), cy + int(p[1]*ch)))
            if len(chk_pts) >= 3:
                colori = []
                try:

                    import numpy as _np
                    with _mss.mss() as sct:
                        monitor = {"top": cy, "left": cx, "width": cw, "height": ch}
                        sct_img = _np.array(sct.grab(monitor))
                        for px, py in chk_pts:
                            y_idx = max(0, min(ch - 1, py - cy))

                            b, g, r, _ = sct_img[y_idx, x_idx]
                            colori.append((int(r), int(g), int(b)))
                except Exception:
                    for px, py in chk_pts:
                        try:
                            colori.append(_pag.pixel(px, py))
                        except Exception:
                            colori.append((0, 0, 0))
                
                max_dist = -1
                anomalo_idx = 0
                for i in range(len(colori)):
                    dist_sum = 0
                    for j in range(len(colori)):
                        if i != j:
                            dist_sum += abs(colori[i][0]-colori[j][0]) + abs(colori[i][1]-colori[j][1]) + abs(colori[i][2]-colori[j][2])
                    if dist_sum > max_dist:
                        max_dist = dist_sum

                print(f"[Anomalia] Colori: {colori} -> Scelto bottone {anomalo_idx}", flush=True)
                _click_hold(btn_pts[anomalo_idx][0], btn_pts[anomalo_idx][1], durata)
            _time.sleep(attesa)

                    if len(roi_poly) >= 3:
                        xs = [p[0] for p in roi_poly]

                                sx = m_left + int((x1 + x2) / 2)

                                    n_v = len(roi_poly)

                                        xi, yi = roi_poly[i_v]
                                        xj, yj = roi_poly[j_v]
                                        if ((yi > ry_rel) != (yj > ry_rel)) and \

                                if inside:

                        except Exception:
                            pass

            if not found:
                print(f"[yolo_drag] Oggetto non trovato (modello: {_model_path})")
            _time.sleep(attesa)

        elif tipo == 'yolo_drag_all':

            try:
                from ultralytics import YOLO as _YOLO

                _YOLO_OK = True

                            img_bgr = sct_img[:, :, :3]
                            results = model(img_bgr, conf=0.7, verbose=False)
                            boxes = results[0].boxes
                            
                            valid_boxes = []
                            for box in boxes:

def run_task(ctx):
    if not _OK:
        return False

    hwnd = ctx.get('hwnd') if ctx else None
    if hwnd is None:
        nome = (ctx or {}).get('finestra', FINESTRA_DEFAULT)
        hwnd = _win32gui.FindWindow(None, nome)
        if not hwnd:
            print(f"[{TASK_META['nome']}] finestra non trovata: {nome}")
            return False
        try:
            _win32gui.SetForegroundWindow(hwnd)
            _time.sleep(0.4)
        except Exception:
            pass

    if not AZIONI:
        print(f"[{TASK_META['nome']}] nessuna azione registrata.")
        return True

    return esegui_azioni(AZIONI, hwnd, current_step=TASK_META.get('step', 0))
